    hidden,

    total_padded_tokens,

    threads = 256
    packed_shape = (num_experts, expert_block_m, hidden)

    def kernel(

        group_padded_offsets: T.Tensor((num_experts + 1,), T.int32),
        packed_input: T.Tensor(packed_shape, T.float16),
    ):

            num_experts,

        ) as (expert_id, bn):
