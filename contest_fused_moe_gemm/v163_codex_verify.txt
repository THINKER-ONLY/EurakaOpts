def _pack_experts_kernel(
    hidden,
    num_experts,
    total_padded_tokens,
    expert_block_m,
):
    block_n = 256
    threads = 256
    packed_shape = (num_experts, expert_block_m, hidden)

    @T.prim_func
    def kernel(
        stacked_expert_tokens: T.Tensor((total_padded_tokens, hidden), T.float16),
        group_sizes: T.Tensor((num_experts,), T.int32),
        group_padded_offsets: T.Tensor((num_experts + 1,), T.int32),
        packed_input: T.Tensor(packed_shape, T.float16),
    ):
        with T.Kernel(
            num_experts,
            T.ceildiv(hidden, block_n),
            threads=threads,
        ) as (expert_id, bn):

            padded_start = group_padded_offsets[expert_id]

                    packed_input[expert_id, i, bn * block_n + j] = (
                        stacked_expert_tokens[padded_start + i, bn * block_n + j]
                    )

    return kernel

    block_n = 128
    block_k = 64
    threads = 256

    @T.prim_func
    def kernel(

        down_w: T.Tensor((64, hidden, 2048), T.float16),
        routed_expert_weights: T.Tensor((total_valid_tokens,), T.float32),
        group_sizes: T.Tensor((64,), T.int32),
        group_offsets: T.Tensor((65,), T.int32),
        group_padded_offsets: T.Tensor((65,), T.int32),
        out: T.Tensor((total_padded_tokens, hidden), T.float16),
    ):

            row_base = 128 + bm * block_m
            group_size = group_sizes[expert_id]
            if row_base < group_size:
                input_shared = T.alloc_shared((block_m, block_k), T.float16)
                weight_shared = T.alloc_shared((block_n, block_k), T.float16)
                output_local = T.alloc_fragment((block_m, block_n), T.float32)
                T.clear(output_local)
                for ko in T.Pipelined(T.ceildiv(2048, block_k), num_stages=2):
                    T.copy(
                        activation[expert_id, row_base : row_base + block_m, ko * block_k : (ko + 1) * block_k],
                        input_shared,
                    )
                    T.copy(
                        down_w[expert_id, bn * block_n : (bn + 1) * block_n, ko * block_k : (ko + 1) * block_k],
                        weight_shared,
                    )
                    T.gemm(input_shared, weight_shared, output_local, transpose_B=True)
                raw_start = group_offsets[expert_id]
                padded_start = group_padded_offsets[expert_id]
                for i, j in T.Parallel(block_m, block_n):
                    if row_base + i < group_size:
                        out[padded_start + row_base + i, bn * block_n + j] = (
                            output_local[i, j]
                            * routed_expert_weights[raw_start + row_base + i]
                        )

    return kernel

        down_w: T.Tensor((64, hidden, 2048), T.float16),
        routed_expert_weights: T.Tensor((total_valid_tokens,), T.float32),
        group_sizes: T.Tensor((64,), T.int32),
        group_offsets: T.Tensor((65,), T.int32),

        out: T.Tensor((total_padded_tokens, hidden), T.float16),

                input_shared = T.alloc_shared((block_m, block_k), T.float16)
                weight_shared = T.alloc_shared((block_n, block_k), T.float16)
                output_local = T.alloc_fragment((block_m, block_n), T.float32)
                T.clear(output_local)

                        input_shared,

                        weight_shared,
                    )
                    T.gemm(input_shared, weight_shared, output_local, transpose_B=True)
                raw_start = group_offsets[expert_id]

**NOVELTY_PROOF**

v163 is not v144 and not v151.

v144 failure mode: all-standard row split still computed the full padded `M192` FC2 path, so it paid for rows `128:192` even when most E64 groups had smaller tails. It saved little real FC2 work.
