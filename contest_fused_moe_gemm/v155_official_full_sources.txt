11: @tilelang.jit(pass_configs={"tl.disable_warp_specialized": True})
12: def moe_forward_tilelang_shared(
13:     d_hidden,
14:     d_expert,
15:     n_shared_experts,
16:     dtype,
17:     num_tokens,
18:     block_token=128,
19:     block_dhidden=128,
20:     block_dexpert=128,
21:     threads=256,
22:     num_stages=1,
23: ):
24:     scale = 1.44269504  # log2(e)
25: 
26:     # Parameters
27:     dhidden = d_hidden
28:     dexpert = d_expert * n_shared_experts
29: 
30:     # Tensors: Note that input shape is reshape to (num_tokens, dhidden)
31:     input_shape = (num_tokens, dhidden)
32:     shared_W_gate_shape = (dexpert, dhidden)
33:     shared_W_up_shape = (dexpert, dhidden)
34:     shared_W_down_shape = (dhidden, dexpert)
35: 
36:     accum_type = T.float32
37: 
38:     @T.prim_func
39:     def kernel_shared(
40:         input: T.Tensor(input_shape, dtype),  # type: ignore
41:         shared_W_gate: T.Tensor(shared_W_gate_shape, dtype),  # type: ignore
42:         shared_W_up: T.Tensor(shared_W_up_shape, dtype),  # type: ignore
43:         shared_W_down: T.Tensor(shared_W_down_shape, dtype),  # type: ignore
44:         up_logits: T.Tensor((num_tokens, dexpert), dtype),  # type: ignore
45:         output: T.Tensor(input_shape, dtype),  # type: ignore
46:     ):
47:         # Step 1: Compute gate and up logits
48:         with T.Kernel(T.ceildiv(num_tokens, block_token), T.ceildiv(dexpert, block_dexpert), threads=threads) as (bx, by):
49:             # Split the block to shared experts and routed experts
50:             input_shared = T.alloc_fragment((block_token, block_dhidden), dtype=dtype)
51:             W_gate_shared = T.alloc_shared((block_dexpert, block_dhidden), dtype=dtype)
52:             W_up_shared = T.alloc_shared((block_dexpert, block_dhidden), dtype=dtype)
53:             # Shared experts: no need to check expert_indices
54: 
55:             gate_logits_local = T.alloc_fragment((block_token, block_dexpert), dtype=accum_type)
56:             up_logits_local = T.alloc_fragment((block_token, block_dexpert), dtype=accum_type)
57: 
58:             T.use_swizzle(10)
59:             T.clear(gate_logits_local)
60:             T.clear(up_logits_local)
61: 
62:             # Parallel for gate and up matmul
63:             for k in T.Pipelined(T.ceildiv(dhidden, block_dhidden), num_stages=num_stages):
64:                 T.copy(input[bx * block_token, k * block_dhidden], input_shared)
65:                 T.copy(shared_W_gate[by * block_dexpert, k * block_dhidden], W_gate_shared)
66:                 T.copy(shared_W_up[by * block_dexpert, k * block_dhidden], W_up_shared)
67:                 T.gemm(input_shared, W_gate_shared, gate_logits_local, transpose_B=True)
68:                 T.gemm(input_shared, W_up_shared, up_logits_local, transpose_B=True)
69: 
70:             # Fuse with SiLU and element-wise product
71:             for i, j in T.Parallel(block_token, block_dexpert):
72:                 gate_logits_local[i, j] = gate_logits_local[i, j] * (1.0 / (1.0 + T.exp2(-gate_logits_local[i, j] * scale)))
73:                 up_logits_local[i, j] = up_logits_local[i, j] * gate_logits_local[i, j]
74: 
75:             T.copy(up_logits_local, up_logits[bx * block_token, by * block_dexpert])
76: 
77:         # Step 2: Compute down logits
78:         with T.Kernel(T.ceildiv(num_tokens, block_token), T.ceildiv(dhidden, block_dhidden), threads=threads) as (bx, by):
79:             up_logits_shared = T.alloc_fragment((block_token, block_dexpert), dtype=dtype)
80:             W_down_shared = T.alloc_shared((block_dhidden, block_dexpert), dtype=dtype)
81:             output_local = T.alloc_fragment((block_token, block_dhidden), dtype=accum_type)
82: 
83:             T.use_swizzle(10)
84:             T.clear(output_local)
85: 
86:             for k in T.Pipelined(T.ceildiv(dexpert, block_dexpert), num_stages=num_stages):
87:                 T.copy(up_logits[bx * block_token, k * block_dexpert], up_logits_shared)
88:                 T.copy(shared_W_down[by * block_dhidden, k * block_dexpert], W_down_shared)
89:                 T.gemm(up_logits_shared, W_down_shared, output_local, transpose_B=True)
90: 
91:             T.copy(output_local, output[bx * block_token, by * block_dhidden])
92: 
93:     return kernel_shared

235: class Expert(nn.Module):
236:     def __init__(self, config: Dict, gate: torch.Tensor, up: torch.Tensor, down: torch.Tensor, d_expert: Optional[int] = None):
237:         super().__init__()
238:         self.config = config
239:         self.act_fn = nn.SiLU()
240:         self.d_hidden: int = config["d_hidden"]
241:         self.d_expert: int = config["d_expert"] if d_expert is None else d_expert
242:         self.device = torch.device("cuda")
243: 
244:         self.W_gate_weight = gate.t().contiguous().to(self.device)
245:         self.W_up_weight = up.t().contiguous().to(self.device)
246:         self.W_down_weight = down.t().contiguous().to(self.device)
247: 
248:     def forward(self, x: torch.Tensor) -> torch.Tensor:
249:         gate = self.act_fn(x @ self.W_gate_weight)
250:         out = (gate * (x @ self.W_up_weight)) @ self.W_down_weight
251:         return out

256:         super().__init__()

264:         logits = x @ self.W_g_weight

336:     @torch.no_grad()
337:     def forward(self, x: torch.Tensor) -> torch.Tensor:
338:         orig_shape = x.shape
339:         batch_size, seq_len, hidden_dim = x.shape
340:         expert_indices, expert_scores = self.gating_network(x)
341:         flat_expert_indices = expert_indices.view(-1)
342:         flat_expert_weights = expert_scores.view(-1)
343:         x_flat = x.view(-1, hidden_dim)
344: 
345:         # Prepare for grouped GEMM
346:         idxs = flat_expert_indices.argsort()
347:         counts = flat_expert_indices.bincount().cpu().numpy()
348:         # counts = flat_expert_indices.bincount()
349:         tokens_per_expert = counts.cumsum()
350:         # tokens_per_expert = torch.cumsum(counts, dim=0)
351:         num_per_tok = self.config["n_experts_per_token"]
352:         token_idxs = idxs // num_per_tok
353: 
354:         # Get stacked expert tokens and expert weights
355: 
356:         for expert_id, end_idx in enumerate(tokens_per_expert):
357:             start_idx = 0 if expert_id == 0 else tokens_per_expert[expert_id - 1]
358:             if start_idx == end_idx:
359:                 continue
360: 
361:             exp_token_idxs = token_idxs[start_idx:end_idx]
362:             expert_tokens = x_flat[exp_token_idxs]
363: 
364:             self.stacked_expert_tokens[start_idx:end_idx] = expert_tokens
365:             self.stacked_expert_tokens_idxs[start_idx:end_idx] = exp_token_idxs
366:             self.stacked_expert_weights[start_idx:end_idx] = flat_expert_weights[idxs[start_idx:end_idx]]
367: 
368:         group_sizes = torch.tensor(counts, dtype=torch.int32, device=self.device)
369:         group_offset = torch.tensor(tokens_per_expert - counts, dtype=torch.int32, device=self.device)
370: 
371:         group_padded_offsets = [0 for _ in range(len(group_sizes))]
372:         for i in range(1, len(group_sizes)):
373:             group_padded_offsets[i] = group_padded_offsets[i - 1] + math.ceil((counts[i - 1] + 1) / self.padding_M) * self.padding_M
374: 
375:         block_token = 128
376:         M = (
377:             math.ceil(self.config["batch_size"] * self.config["seq_len"] * self.config["n_experts_per_token"] / block_token)
378:             + self.config["n_routed_experts"]
379:         )
380:         group_idx_for_bx = [0 for _ in range(M)]
381: 
382:         for bx in range(M):
383:             m_start_padded = bx * block_token
384:             for i in range(self.config["n_routed_experts"]):
385:                 if m_start_padded >= group_padded_offsets[i]:
386:                     group_idx_for_bx[bx] = i
387: 
388:         group_padded_offsets = torch.tensor(group_padded_offsets, dtype=torch.int32, device=self.device)
389:         group_idx_for_bx = torch.tensor(group_idx_for_bx, dtype=torch.int32, device=self.device)
390: 
391:         # Multi-stream execution
392:         shared_stream = torch.cuda.Stream()
393:         routed_stream = torch.cuda.default_stream()
394:         torch.cuda.synchronize()
395: 
396:         with torch.cuda.stream(routed_stream):
397:             # Tilelang version: Grouped GEMM
398:             self.routed_kernel(
399:                 self.stacked_expert_tokens,
400:                 self.stacked_expert_w_gate,
401:                 self.stacked_expert_w_up,
402:                 self.stacked_expert_w_down,
403:                 self.stacked_expert_weights,
404:                 group_sizes,
405:                 group_offset,
406:                 group_padded_offsets,
407:                 group_idx_for_bx,
408:                 self.up_logits_routed,
409:                 self.expert_output_routed,
410:             )
411: 
412:             # Scatter reduce
413:             self.expert_cache = torch.scatter_reduce(
414:                 self.expert_cache,
415:                 0,
416:                 self.stacked_expert_tokens_idxs.view(-1, 1).repeat(1, x_flat.shape[-1]),
417:                 self.expert_output_routed,
418:                 reduce="sum",
419:             )
420:             routed_output = self.expert_cache.view(*orig_shape)
421: 
422:         with torch.cuda.stream(shared_stream):
423:             self.shared_kernel(
424:                 x_flat,
425:                 self.shared_expert.W_gate_weight,
426:                 self.shared_expert.W_up_weight,
427:                 self.shared_expert.W_down_weight,
428:                 self.up_logits_shared,
429:                 self.expert_output_shared,
430:             )
431:             shared_output = self.expert_output_shared.view(*orig_shape)
432: 
433:         torch.cuda.synchronize()
434: 
435:         return shared_output + routed_output

26: def grouped_gemm_ptr(batch_sizes_list, K, N, block_M, block_N, block_K, num_stages=2, threads=128, dtype=T.float16):
27:     # Keep per-group tensors separate and pass them via pointer tables.
28:     # We currently use a common max_M storage shape per group because
29:     # ptr-backed tensors with runtime-varying shapes are not stable enough yet.
30:     # Multi-stage software pipelining on ptr-backed tensors is not correct yet.
31:     # Keep a single-stage pipeline so the ptr path can still use T.copy lowering.
32:     copy_num_stages = 1
33:     batch_count = len(batch_sizes_list)
34:     max_M = max(batch_sizes_list)
35:     batch_tile_offsets = [0]
36:     for size in batch_sizes_list[:-1]:
37:         batch_tile_offsets.append(batch_tile_offsets[-1] + math.ceil(size / block_M))
38:     total_m_blocks = sum(math.ceil(size / block_M) for size in batch_sizes_list)
39:     accum_dtype = T.float32
40: 
41:     @T.prim_func
42:     def kernel(
43:         A_ptrs: T.Tensor([batch_count], T.ptr),
44:         B_ptrs: T.Tensor([batch_count], T.ptr),
45:         C_ptrs: T.Tensor([batch_count], T.ptr),
46:         batch_tile_offsets: T.Tensor([batch_count], T.int32),
47:     ):
48:         with T.Kernel(total_m_blocks, T.ceildiv(N, block_N), threads=threads) as (bx, by):
49:             A_shared = T.alloc_shared((block_M, block_K), dtype)
50:             B_shared = T.alloc_shared((block_K, block_N), dtype)
51:             C_local = T.alloc_fragment((block_M, block_N), accum_dtype)
52:             cur_batch_idx = T.alloc_var(dtype=T.int32)
53:             cur_tile_offset = T.alloc_var(dtype=T.int32)
54: 
55:             cur_batch_idx = 0
56:             cur_tile_offset = 0
57:             for i in range(batch_count):
58:                 in_cur_batch_idx = bx >= batch_tile_offsets[i]
59:                 cur_batch_idx = T.if_then_else(in_cur_batch_idx, i, cur_batch_idx)
60:                 cur_tile_offset = T.if_then_else(in_cur_batch_idx, batch_tile_offsets[i], cur_tile_offset)
61: 
62:             m_start = (bx - cur_tile_offset) * block_M
63:             A = T.make_tensor(A_ptrs[cur_batch_idx], (max_M, K), dtype)
64:             B = T.make_tensor(B_ptrs[cur_batch_idx], (K, N), dtype)
65:             C = T.make_tensor(C_ptrs[cur_batch_idx], (max_M, N), dtype)
66: 
67:             T.clear(C_local)
68:             for ko in T.Pipelined(T.ceildiv(K, block_K), num_stages=copy_num_stages):
69:                 T.copy(A[m_start, ko * block_K], A_shared)
70:                 T.copy(B[ko * block_K, by * block_N], B_shared)
71:                 T.gemm(A_shared, B_shared, C_local)
72: 
73:             T.copy(C_local, C[m_start, by * block_N])
74: 
75:     return kernel
