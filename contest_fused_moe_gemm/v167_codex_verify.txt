def _swiglu_kernel(
    intermediate,
    num_experts,
    expert_block_m,
):
    block_n = 256
    threads = 256
    shape = (num_experts, expert_block_m, intermediate)

    @T.prim_func
    def kernel(
        gate_output: T.Tensor(shape, T.float16),
        up_output: T.Tensor(shape, T.float16),
        group_sizes: T.Tensor((num_experts,), T.int32),
        activation: T.Tensor(shape, T.float16),
    ):
        with T.Kernel(
            num_experts,
            T.ceildiv(intermediate, block_n),
            threads=threads,
        ) as (expert_id, bn):
            group_size = group_sizes[expert_id]
            for i, j in T.Parallel(expert_block_m, block_n):
                if i < group_size:
                    gate = gate_output[expert_id, i, bn * block_n + j]
                    activation[expert_id, i, bn * block_n + j] = (
                        gate
                        * T.sigmoid(gate)
                        * up_output[expert_id, i, bn * block_n + j]
                    )

    return kernel

def _swiglu_kernel(
    intermediate,
    num_experts,
    expert_block_m,
):
    block_n = 256
    threads = 256
    shape = (num_experts, expert_block_m, intermediate)

    @T.prim_func
    def kernel(
        gate_output: T.Tensor(shape, T.float16),
        up_output: T.Tensor(shape, T.float16),
        group_sizes: T.Tensor((num_experts,), T.int32),
        activation: T.Tensor(shape, T.float16),
    ):
        with T.Kernel(
            num_experts,
            T.ceildiv(intermediate, block_n),
            threads=threads,
        ) as (expert_id, bn):
            group_size = group_sizes[expert_id]
            for i, j in T.Parallel(expert_block_m, block_n):
                if i < group_size:
                    gate = gate_output[expert_id, i, bn * block_n + j]
                    activation[expert_id, i, bn * block_n + j] = (
                        gate
                        * T.sigmoid(gate)
                        * up_output[expert_id, i, bn * block_n + j]
                    )

    return kernel
