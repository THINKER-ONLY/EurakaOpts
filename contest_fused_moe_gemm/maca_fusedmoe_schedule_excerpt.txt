38:     @T.prim_func
39:     def kernel_shared(
40:         input: T.Tensor(input_shape, dtype),  # type: ignore
41:         shared_W_gate: T.Tensor(shared_W_gate_shape, dtype),  # type: ignore
42:         shared_W_up: T.Tensor(shared_W_up_shape, dtype),  # type: ignore
43:         shared_W_down: T.Tensor(shared_W_down_shape, dtype),  # type: ignore
44:         up_logits: T.Tensor((num_tokens, dexpert), dtype),  # type: ignore
45:         output: T.Tensor(input_shape, dtype),  # type: ignore
46:     ):
47:         # Step 1: Compute gate and up logits
48:         with T.Kernel(T.ceildiv(num_tokens, block_token), T.ceildiv(dexpert, block_dexpert), threads=threads) as (bx, by):
49:             # Split the block to shared experts and routed experts
50:             input_shared = T.alloc_fragment((block_token, block_dhidden), dtype=dtype)
51:             W_gate_shared = T.alloc_shared((block_dexpert, block_dhidden), dtype=dtype)
52:             W_up_shared = T.alloc_shared((block_dexpert, block_dhidden), dtype=dtype)
53:             # Shared experts: no need to check expert_indices
54: 
55:             gate_logits_local = T.alloc_fragment((block_token, block_dexpert), dtype=accum_type)
56:             up_logits_local = T.alloc_fragment((block_token, block_dexpert), dtype=accum_type)
57: 
58:             T.use_swizzle(10)
59:             T.clear(gate_logits_local)
60:             T.clear(up_logits_local)
61: 
62:             # Parallel for gate and up matmul
63:             for k in T.Pipelined(T.ceildiv(dhidden, block_dhidden), num_stages=num_stages):
64:                 T.copy(input[bx * block_token, k * block_dhidden], input_shared)
65:                 T.copy(shared_W_gate[by * block_dexpert, k * block_dhidden], W_gate_shared)
66:                 T.copy(shared_W_up[by * block_dexpert, k * block_dhidden], W_up_shared)
67:                 T.gemm(input_shared, W_gate_shared, gate_logits_local, transpose_B=True)
68:                 T.gemm(input_shared, W_up_shared, up_logits_local, transpose_B=True)
69: 
70:             # Fuse with SiLU and element-wise product
71:             for i, j in T.Parallel(block_token, block_dexpert):
72:                 gate_logits_local[i, j] = gate_logits_local[i, j] * (1.0 / (1.0 + T.exp2(-gate_logits_local[i, j] * scale)))
73:                 up_logits_local[i, j] = up_logits_local[i, j] * gate_logits_local[i, j]
74: 
75:             T.copy(up_logits_local, up_logits[bx * block_token, by * block_dexpert])
76: 
77:         # Step 2: Compute down logits
78:         with T.Kernel(T.ceildiv(num_tokens, block_token), T.ceildiv(dhidden, block_dhidden), threads=threads) as (bx, by):
79:             up_logits_shared = T.alloc_fragment((block_token, block_dexpert), dtype=dtype)
80:             W_down_shared = T.alloc_shared((block_dhidden, block_dexpert), dtype=dtype)
81:             output_local = T.alloc_fragment((block_token, block_dhidden), dtype=accum_type)
82: 
83:             T.use_swizzle(10)
84:             T.clear(output_local)
85: 
86:             for k in T.Pipelined(T.ceildiv(dexpert, block_dexpert), num_stages=num_stages):
87:                 T.copy(up_logits[bx * block_token, k * block_dexpert], up_logits_shared)
88:                 T.copy(shared_W_down[by * block_dhidden, k * block_dexpert], W_down_shared)
89:                 T.gemm(up_logits_shared, W_down_shared, output_local, transpose_B=True)
90: 
91:             T.copy(output_local, output[bx * block_token, by * block_dhidden])
92: 
93:     return kernel_shared
